# Brain Guardian — crawler policy
#
# /guardian is the public acquisition landing page — allow all crawlers to index it.
# Its render assets (JS/CSS bundles, favicon, and preview image) must also be
# reachable so crawlers can fully render the page and generate link previews.
# llms.txt and sitemap.xml are always allowed for all crawlers.
# Everything else is a private care workspace and must not be indexed or crawled.

Allow: /guardian$
Allow: /guardian/
Allow: /assets/
Allow: /opengraph.jpg
Allow: /favicon.svg
Allow: /llms.txt
Allow: /sitemap.xml
Disallow: /guardian/success
